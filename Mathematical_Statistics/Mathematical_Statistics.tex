\chapter{基本概念与统计模型}

\section{基本概念}

\begin{definition}
    \textbf{总体}由与研究问题有关的所有个体组成。样本中的个体数目称为\textbf{样本大小}，抽取样本的行为称为\textbf{抽样}。

    根据总体个数有限与无限，分为有限总体和无限总体。
\end{definition}

总体也可看作所有个体上某种数量指标的集合，可以用一个随机变量及其概率分布（总体分布）描述。当数量指标有多个时，用随机向量表示总体。

\begin{definition}
    样本$\mathbf{X}=(X_1,\cdots,X_n)$可能取值的全体，构成\textbf{样本空间}，记为$\mathcal{X}$。
\end{definition}

样本的两重性：既可看做随机变量（随机向量）（抽样前，Uppercase），又可看作确定的数（抽样后，lowercase）。

简单随机抽样：(1)代表性：等概率抽选；(2)独立性：样本个体值不影响其他值。

\begin{definition}[简单随机样本]
    总体（分布）F中的容量为n的样本样本$X_1,\cdots,X_n$i.i.d.~F称为\textbf{简单随机样本}（简单样本/随机样本）。
\end{definition}

\section{统计推断}

\begin{definition}
    将出现在样本分布中的未知常数称为\textbf{参数（向量）}，如$N(a,\sigma^2)\to (a,\sigma^2)$.

    参数的取值范围称为\textbf{参数空间}。
\end{definition}

\begin{definition}
    样本分布包含未知参数时，参数取不同值的样本分布构成分部族：
    $$\mathcal{F} =\{F_{\boldsymbol{\theta}},\boldsymbol{\theta}\in\Theta\}\text{或}\mathcal{F} =\{f_{\boldsymbol{\theta}},\boldsymbol{\theta}\in\Theta\}$$
    其中$\Theta\in\mathbb{R}^k$为参数空间。
\end{definition}

\begin{example}
    指数分布$\exp(\lambda)$的样本分布族为
    $$\mathcal{F} =\{f_{\lambda}(x)=f(x,\lambda):\lambda>0\}$$
\end{example}

\begin{definition}
    从总体中抽取一定大小的样本去推断总体的概率分布的方法称为\textbf{统计推断}(statistical inference)。当样本分布形式已知，而含有未知实参数时的统计推断问题称为\textbf{参数统计推断问题}，否则称为非参数统计推断问题。
\end{definition}

参数统计推断有多种，例如参数估计和假设检验。参数估计问题通常分为两类：\textbf{点估计}和\textbf{区间估计}，前者估计参数（或参数的函数）的值，后者给出取值范围。

\section{统计量}

\begin{definition}
    由样本算出的量（样本的函数）称为\textbf{统计量}(statistic)。
\end{definition}

\begin{remark}
    \begin{itemize}
        \item 统计量只允许与样本有关而不允许与未知参数有关
        \item 统计量也具有两重性
    \end{itemize}
\end{remark}

\subsection{常见统计量}

\begin{definition}
    设$X_1,\cdots,X_n$是从某总体X中抽取的样本，称
    $$\bar{X}=\frac{1}{n}\sum_{i=1}^n X_i$$
    为\textbf{样本均值}(sample mean). 称
    $$S^2=\frac{1}{n-1}\sum_{i=1}^n (X_i-\bar{X})^2$$
    为\textbf{样本方差}(sample variance).称
    $$a_{n,k}=\frac{1}{n}\sum_{i=1}^n X_i^k,k=1,2,\cdots$$
    为\textbf{样本k阶原点矩}，$a_{n,1}=\bar{X}$。称
    $$m_{n,k}=\frac{1}{n}\sum_{i=1}^n (X_i-\bar{X})^k$$
    为\textbf{样本k阶中心距}。
\end{definition}

\begin{proposition}
    (1)$\sum_{i=1}^n (X_i-\bar{X})=0$

    (2)设$a,b$为常数，$Y_i=aX_i+b$，则
    $$\bar{Y}=a\bar{X}+b,S_Y^2=a^2 S_X^2$$

    (3)$\forall \text{常数}c$，有
    $$\sum_{i=1}^n(X_i-c)^2\ge\sum_{i=1}^n (X_i-\bar{X})^2$$
    即均值是偏差平方最小准则下的最佳估计
\end{proposition}

\begin{definition}[二维随机向量的样本矩]
    设$(X_1,Y_1),\cdots,(X_n,Y_n)$为二维样本$F(x,y)$中抽取的样本，则已定义$\bar{X},\bar{Y},S^2_X,S_Y^2$.
    $$S_{XY}=\frac{1}{n}(X_i-\bar{X})(Y_i-\bar{Y})$$
    称为X和Y的\textbf{样本协方差}(sample covariance).
    $$r=\frac{\sum_{i=1}^n (X_i-\bar{X})(Y_i-\bar{Y})}{\sqrt{[(n-1)S_X^2]\cdot [(n-1)S_Y^2]}}$$
    称为\textbf{样本相关系数}(sample covariance coefficient).
\end{definition}

\begin{definition}
    设$X_1,\cdots,X_n$是从某总体F中抽取的样本，按大小排序为$X_{(1)},\cdots,X_{(n)}$，称$(X_{(1)},\cdots,X_{(n)})$是样本$(X_1,\cdots,X_n)$的\textbf{次序统计量}(order statistics).$(X_{(1)},\cdots,X_{(n)})$的一部分也可称为次序统计量。
\end{definition}

\begin{definition}
    \begin{itemize}
        \item \textbf{样本中位数}(sample median)：$$m_{\frac{1}{2}}=\begin{cases}
        X_{((n+1)/2)},n\text{为奇数}\\
        \dfrac{1}{2}[X_{(n/2)}+X_{(n/2+1)}],n\text{为偶数}
        \end{cases}$$
        \item \textbf{样本p分位数}$(0<p<1)$(sample p-fractile)：$$X_{(m)},m=[(n+1)p]$$
        \item \textbf{样本极值}(extremum of sample)：$X_{(1)},X_{(n)}$
        \item \textbf{样本极差}(sample range)：$R=X_{(n)}-X_{(1)}$反映分散程度
    \end{itemize}
\end{definition}

\begin{definition}
    \textbf{样本变异系数}(sample coefficient of variation)与\textbf{总体变异系数}(population coefficient of variation)：$$\hat{\nu}=\frac{S_n}{\bar{Xt}},\nu=\frac{\sqrt{\mathrm{Var}(X)}}{\mathbb{E} (X)}$$
    后者以总体均值为单位度量总体分布的分散程度，前者反映后者的信息。
\end{definition}

\begin{definition}
    \textbf{样本偏度}(sample skewness):
    $$\hat{\beta}_1=\frac{m_{n,3}}{m_{n,2}^{3/2}}=\frac{\sqrt{n}\sum_{i=1}^{n}(X_i-\bar{X})^3}{\left(\sum_{i=1}^n(X_i-\bar{X})^2\right)^{3/2}}$$
    \textbf{总体偏度}(population skewness):
    $$\beta_1=\frac{\mu_3}{\mu_2^{3/2}}$$
    其中$\mu_k$为总体的k阶中心矩。

    后者衡量总体分布的非对称性，前者反映后者的信息。
\end{definition}

\begin{definition}
    \textbf{样本峰度}(sample kurtosis):
    $$\hat{\beta}_2=\frac{m_{n,4}}{m_{n,2}^{2}}-3=\frac{n\sum_{i=1}^{n}(X_i-\bar{X})^4}{\left(\sum_{i=1}^n(X_i-\bar{X})^2\right)^{2}}-3$$
    \textbf{总体峰度}(population kurtosis):
    $$\beta_2=\frac{\mu_4}{\mu_2^{2}}-3$$
    其中$\mu_k$为总体的k阶中心矩。

    后者衡量概率密度函数在众数（最大值）附近尖锐程度，前者反映后者的信息。
\end{definition}

\subsection{经验分布函数}

\begin{definition}
    设有$F(x)$的次序统计量$X_{(1)},\cdots,X_{(n)}$，称函数
    $$F_n(x)=\begin{cases}
        0,x<X_{(1)}\\
        \dfrac{k}{n},X_{(k)}\le x <X_{(k+1)},k=1,2,\cdots,n-1\\
        1,X_{(n)}\le x
    \end{cases}$$
    为\textbf{经验分布函数}(empirical distribution function).
\end{definition}

经验分布函数单调不减、右连续，是$F(x)=\mathbb{P} (X\le x)$的统计量。

定义\textbf{示性函数}
$$I_A(x)=\begin{cases}
    1,x\in A\\
    0,otherwise
\end{cases}$$
则
$$F_n(x)=\frac{1}{n}\sum_{i=1}^n I_{(-\infty,x]}(X_i)$$

在时刻$x$以前，$X_i$发生了吗？
\begin{align*}
    Y_i(x)=I_{(-\infty,x]}(X_i)&\implies P(Y_i(x)=1)=F(x),Y_1,\cdots,Y_n \text{i.i.d.}\sim B(1,F(x))\\
    &\implies nF_n(x)=\sum_{i=1}^n Y_i\sim B(n,F(x))
\end{align*}
由此可得$F_n(x)$的大样本性质：
\begin{theorem}
    (1)由中心极限定理，
    $$\frac{\sqrt{n}(F_n(x)-F(x))}{\sqrt{F(x)(1-F(x))}}\xrightarrow{\mathcal{L}}N(0,1),n\to\infty$$
    其中$\xrightarrow{\mathcal{L}}$表示依分布收敛

    (2)由Bernoulli（或辛钦）大数定律，
    $$F_n(x)\xrightarrow{\mathbb{P}}F(x),n\to\infty$$

    (3)由Borel大数定律，以概率1收敛（a.e.收敛）：
    $$\mathbb{P} (\lim_{n\to\infty}F_n(x)=F(x))=1$$

    (4)(Glivenko-Cantelli定理)用$D_n$衡量$F_n(x)$总体上靠近$F(x)$的程度：
    $$D_n=\sup_{-\infty<x<\infty}|F_n(x)-F(x)|$$
    则有
    $$\mathbb{P} (\lim_{\to\infty}D_n=0)=1$$
\end{theorem}

\section{统计模型}

一个问题的\textbf{统计模型}(statistical model)就是研究该问题时所抽样本的分布，又称概率模型，例如正态（分布）模型、泊松（分布）模型。

\section{指数型分布族}

\begin{definition}
    设$\mathcal{F}=\{f(x,\boldsymbol{\theta}):\boldsymbol{\theta}\in\Theta\}$是定义在样本空间$\mathcal{X} $上的分布族，，若概率密度函数可以表示为以下形式：
    $$f(x,\boldsymbol{\theta})=C(\boldsymbol{\theta})\exp\left[\sum_{i=1}^k Q_i(\boldsymbol{\theta})T_i(x)\right]h(x)$$
    则称此分布族为\textbf{指数型分布族/指数族}(exponential family)，其中$k\in\mathbb{N}^*,C(\boldsymbol{\theta})>0,h(x)>0$.
\end{definition}

\begin{remark}
    指数族中的所有概率密度函数$f(x,\boldsymbol{\theta})$的对$x$的支集必然不依赖于$\boldsymbol{\theta}$.
\end{remark}

\begin{definition}
    如果指数族的概率密度函数有如下形式：
    $$f(x,\boldsymbol{\theta})=C^*(\boldsymbol{\theta})\exp\left[\sum_{i=1}^k\theta_i T_i(x)\right]h(x)$$
    则称它为指数族的\textbf{自然形式}(natural form)，此时集合
    $$\Theta^*=\left\{(\theta_1,\cdots,\theta_k):\int_{\mathcal{X}}\exp\left[\sum_{i=1}^k\theta_i T_i(x)\right]h(x)\,ds<\infty\right\}$$
    称为\textbf{自然参数空间}(natural parametric space).
\end{definition}

\begin{remark}
    其中的参数组$\boldsymbol{\theta}$可能不是原分布中的参数组，而是令$\varphi_i=Q_i(\boldsymbol{\theta})$再改写符号$\boldsymbol{\varphi}$为$\boldsymbol{\theta}$.
\end{remark}

\begin{theorem}
    指数族的自然形式下，自然参数空间为凸集。
\end{theorem}

\begin{theorem}
    设指数族的自然形式下，自然参数空间有内点，记$\Theta_0=\mathrm{int}\Theta^*$.设实函数$g(x)$使得以下积分在$\Theta_0$内存在且有限：
    $$G(\boldsymbol{\theta})=\int_{\mathcal{X} }g(x)\exp\left[\sum_{j=1}^k \theta_j T_j(x)\right]h(x)\,dx$$
    则$G(\boldsymbol{\theta})$的任意偏导数存在且可以与积分号换序。
\end{theorem}

\section{贝叶斯模型}

全概率+条件概率$\implies$
\begin{theorem}[贝叶斯公式(Bayes formula)]
    设$\{B_!,\cdots,B_n\}$是样本空间$\Omega$的完备事件群（分划），$A$为$\Omega$中的事件，且$\mathbb{P}(B_i)>0,\mathbb{P} (A)>0$，则
    $$\mathbb{P}(B_i|A)=\frac{\mathbb{P} (A|B_i)\mathbb{P} (B_i)}{\mathbb{P} (A)}=\frac{\mathbb{P} (A|B_i)\mathbb{P} (B_i)}{\sum_{j=1}^n \mathbb{P}(A|B_j)(B_j)},i=1,\cdots,n$$
\end{theorem}
\begin{remark}
    $\mathbb{P} (B_i)$为A未发生时对时间$B_i$发生可能性大小的估计，而$\mathbb{P} (B_i|A)$是在已知A发生后（新信息），对$B_i$的发生可能性大小的新的认识。

    “结果”推“原因”：已知结果A，估计是哪一个原因导致A发生。
\end{remark}

统计推断使用的三种信息：\begin{itemize}
    \item 总体信息
    \item 样本信息
    \item 先验信息：抽样之前有关统计推断问题中未知参数的一些信息
\end{itemize}
基于前两者：经典统计学；基于全部三者：Bayes统计学。

\begin{definition}
    参数空间$\Theta$上$\theta$的任一概率分布称为\textbf{先验分布}(prior distribution)，记为$\pi(\theta)$.
\end{definition}

\begin{definition}
    在获得样本$\mathbf{X}=\mathbf{x}$后，$\theta$的\textbf{后验分布}(posterior distribution)记为$\pi(\theta|\mathbf{x})$，满足
    $$\pi(\theta|\mathbf{x})=\frac{f(\mathbf{x},\theta)}{m(\mathbf{x})}=\frac{f(\mathbf{x}|\theta)\pi(\theta)}{\int_{\Theta}f(\mathbf{x}|\theta)\pi(\theta)\,d\theta}$$
    其中$f(\mathbf{x}|\theta)$为样本$\mathbf{x}$的条件概率密度函数，$f(\mathbf{x},\theta)$为$\mathbf{X},\theta$的联合概率密度函数，$m(\mathbf{x})$是$\mathbf{X}$的边缘分布。离散情形下，这个公式就是Bayes公式。
\end{definition}
先验分布是抽取样本$\mathbf{X}$之前对参数$\theta$的认识，后验分布是抽取样本之后对$\theta$的认识。

Bayes学派：对参数$\theta$的统计推断必须基于$\theta$的后验分布。

先验分布的获取：\begin{itemize}
    \item 主观概率
    \item 无信息先验
    \item 共轭先验
\end{itemize}

\subsection{无信息先验}

$\Theta$有界时，无信息先验即均匀分布；$\Theta$无界时可以使用广义先验分布：

\begin{definition}
    设随机变量$X\sim f(x|\theta),\theta\in\Theta$，若$\theta$的先验密度$\pi(\theta)$满足\begin{enumerate}
        \item $\pi(\theta)\ge 0$且$\int_{\Theta}\pi(\theta)\,d\theta=\infty$
        \item 后验密度$\pi(\theta|x)$是概率密度函数
        \end{enumerate}
        则称$\pi(\theta)$为$\theta$的\textbf{广义（无信息）先验密度}(improper prior density).
\end{definition}

常取$\pi(\theta)\equiv 1$（乘以正常数无影响），此时
$$\pi(\theta|\mathbf{x})=\frac{f(\mathbf{x}|\theta)}{\int_{\Theta}f(\mathbf{x}|\theta)\,d\theta}$$

\textbf{位置参数族}：概率密度函数形如$f(x-\theta)$，$\theta$为位置参数。认为$X$与$Y=X+c$有相同的统计问题结构，于是有相同的无信息先验，此时$\pi(\theta)\equiv 1$.

\textbf{刻度参数族}：总体分布族的密度函数形如$\sigma^{-1}\varphi(x/\sigma)$，$\sigma>0$为刻度参数认为$X$与$Y=cX(c>0)$有相同的统计问题结构，于是有相同的无信息先验，此时$\pi(\sigma)=1/\sigma$.

\subsection{共轭先验}

\begin{definition}
    设$\mathcal{F}$表示又$\theta$的先验分布$\pi(\theta)$构成的分布族，如果$\forall \pi\in\mathcal{F} $及样本$\mathbf{x}$，后验分布$\pi(\theta|\mathbf{x})\in\mathcal{F}$，则称$\mathcal{F}$是一个\textbf{共轭先验分布族}(conjugate prior distribution family).
\end{definition}

共轭先验分布易于计算，并且其后验分布中一些参数可以得到很好的解释。

\chapter{抽样分布和数据的简化}

\begin{definition}
    \textbf{统计量}是样本的函数，原则上可由样本分布推出其概率分布，称之为\textbf{抽样分布}。
\end{definition}

给出抽样分布的封闭表达式，称为\textbf{精确抽样分布}，但通常难以求出，且躲在正态分布条件下得到；多数情况下研究\textbf{极限分布}，样本量大时近似精确分布。

好的统计量能够简化数据，且信息无损失，此时称之为\textbf{充分统计量}。

\section{正态总体的样本均值和样本方差的分布}

\subsection{正态变量的线性组合}

\begin{theorem}
    设$X_1,\cdots,X_n$ i.i.d.$\sim N(a_k,\sigma_k^2)$，$c_1,\cdots,c_k$为常数，记$T=\sum_{k=1}^n c_k X_k$，则$T\sim N(\mu,\tau^2)$，其中$\mu=\sum_{k=1}^n c_k a_k,\tau^2=\sum_{k=1}^n c_k^2 \sigma^2_k$.
\end{theorem}

\begin{corollary}
    设$a_1=\cdots=a_n=a,\sigma_1=\cdots=\sigma_n=\sigma$，则
    $$T\sim N\left(a\sum_{k=1}^n c_k^2,\sigma^2\sum_{k=1}^n c_k^2\right)$$
    令$c_1=\cdots=c_n=1/n$，则
    $$T=\bar{X}\sim N\left(a,\frac{\sigma^2}{n}\right)$$
\end{corollary}

\begin{theorem}
    设$X_1,\cdots,X_n$ i.i.d.$\sim N(a,\sigma^2)$，$\mathbf{Y}=A\mathbf{X}$，即
    $$\begin{bmatrix}
        Y_1\\
        \vdots\\
        Y_n
    \end{bmatrix}=\begin{bmatrix}
        a_{11} & \cdots & a_{1n}\\
        \vdots & & \vdots \\
        a_{n1} & \cdots & a_{nn}
    \end{bmatrix}\begin{bmatrix}
        X_1\\
        \vdots\\
        X_n
    \end{bmatrix}$$
    则\begin{enumerate}
        \item $\mathbf{Y}$联合高斯，且
        $$\E[Y_i]=a\sum_{k=1}^n a_{ik},\quad \Var(Y_i)=\sigma^2\sum_{k=1}^n a_{ik}^2,\quad \Cov(Y_i,Y_j)=\sigma^2\sum_{k=1}^n a_{ik}a_{jk}$$
        \item 特别地，如果$A$是单位正交阵，则$Y_1,\cdots,Y_n$ i.i.d.$\sim N(\mu_i,\sigma^2)$，其中$\mu_i=a\sum_{k=1}^n a_{ik}$.进一步，$a=0$时$Y_1,\cdots,Y_n\iid \sim N(0,\sigma^2)$
    \end{enumerate}
\end{theorem}

\begin{theorem}
    设$X_1,\cdots,X_n$ i.i.d.$\sim N(a,\sigma^2)$，则\begin{enumerate}
        \item $\bar{X}\sim N(a,\sigma^2)$
        \item $\dfrac{(n-1)S^2}{\sigma^2}\sim \chi_{n-1}^2$
        \item $\bar{X}$与$S^2$独立
    \end{enumerate}
\end{theorem}
\begin{proof}
    1.显然。对2，令单位正交阵
    $$A=\begin{bmatrix}
        \dfrac{1}{\sqrt{n}} & \cdots & \dfrac{1}{\sqrt{n}}\\
        a_{21} & \cdots & a_{2n}\\
        \vdots & & \vdots\\
        a_{n1} & \cdots & a_{nn}
    \end{bmatrix}$$
    则有$$Y_1=\frac{1}{\sqrt{n}}\sum_{k=1}^n X_k=\sqrt{n}\bar{X}$$
    $$\sum_{k=1}^n Y_k^2=\sum_{k=1}^n X_k^2$$
    因此
    $$(n-1)S^2=\sum_{k=1}^n (X_k-\bar{X})^2=\sum_{k=1}^n X_k^2-n\bar{X}^2=\sum_{k=1}^n Y_k^2-Y_1^2=\sum_{k=2}^n Y_k^2$$
    而$Y_1,\cdots,Y_n\iid\sim N(\mu_k,\sigma^2)$，其中$\mu_i=a\sum_{k=1}^n a_{ik}$，根据$A$的行向量的正交性知$\mu_k=0$，因此
    $$\frac{(n-1)S^2}{\sigma^2}=\sum_{k=2}^n \left(\frac{Y_k}{\sigma}\right)^2\sim \chi_{n-1}^2$$

    对3，$\bar{X}$只与$Y_1$有关，$S^2$只与$Y_2,\cdots,Y_n$有关，得证。
\end{proof}

\section{卡方分布，t分布和F分布}

\subsection{卡方分布}

\begin{definition}
    设$X_1,\cdots,X_n$ i.i.d.$\sim N(0,1)$，称
    $$\xi=\sum_{k=1}^n X_k^2$$
    是自由度为n的\textbf{$\chi^2$变量}，其分布称为自由度为n的\textbf{$\chi^2$分布}。
\end{definition}

\begin{theorem}
    设$\xi\sim \chi_{n}^2$，则其概率密度函数为
    $$g_n(x)=\frac{1}{2^{n/2}\Gamma(n/2)}x^{n/2-1}\e^{-x/2},x>0$$
    即$\chi_{n}^2=\Gamma(\frac{n}{2},\frac{1}{2})$.
\end{theorem}
自由度越大，$\chi_n^2$的概率密度曲线越由“偏左”趋于“对称”（极限为正态分布）；$n=1,2$单调递减，$n\ge 3$有单峰。

将$\chi_n^2$分布的上侧$\alpha(0<\alpha<1)$分位数记为$c=\chi_{n}^2(\alpha)$，它满足$\xi\sim\chi_n^2,\mathbb{P}(\xi>c)=\alpha$，通常取较小的值。

\begin{proposition}[$\chi^2$变量的性质]
    设\rv $\xi\sim\chi_n^2$，则
    \begin{enumerate}
        \item $\xi$的特征函数为$\varphi(t)=(1-2\im t)^{-n/2}$，矩母函数为$\phi(t)=(1-2t)^{-n/2}$.
        \item $\E(\xi)=n,\Var(\xi)=2n$.
        \item 设$Y\sim\Gamma(\alpha,\lambda)$，则$Z=2\lambda Y\sim\chi_{2n}^2$.
        \item 设$Z_i\iid\sim \chi_{n_i}^2$，则$\sum_{i=1}^k Z_i\sim \chi_{n_1+\cdots+n_k}^2$.
    \end{enumerate}
\end{proposition}
\begin{proof}
    利用$\Gamma(n,\lambda)$的矩母函数为$\left(\dfrac{\lambda}{\lambda-t}\right)^n$可得。
\end{proof}

\subsection{t分布}

\begin{definition}
    设$\rv X\sim N(0,1),Y\sim \chi_n^2$且两者独立，称
    $$T=\frac{X}{\sqrt{Y/n}}$$
    为自由度为n的\textbf{t变量}，其分布称为自由度为n的\textbf{t分布},记为$t_n$。
\end{definition}

\begin{theorem}
    设$\rv T\sim t_n$，则其概率密度函数为
    $$t_n(x)=\frac{\Gamma(\frac{n+1}{2})}{\Gamma(\frac{n}{2})\sqrt{n\pi}}\left(1+\frac{x^2}{n}\right)^{-\frac{n+1}{2}}$$
\end{theorem}

$t_n$类似标准正态分布$N(0,1)$，为单峰偶函数，在$x=0$处取最大值，但是峰值相对低、尾部相对粗，$n$越大越接近$N(0,1)$.

将$t_n$分布的双侧上$\alpha(0<\alpha<1)$分位数记为$c=t_n(\alpha)$，它满足$T\sim t_n,\mathbb{P}(T>c)=\alpha$。

\begin{proposition}[$t_n$变量的性质]
    设$\rv T\sim t_n$，则\begin{enumerate}
        \item $\E[T^r]$仅当$r<n$时存在，且
        $$\E[T^r]=\begin{cases}
            n^{r/2}\dfrac{\Gamma(\frac{r+1}{2})\Gamma(\frac{n-r}{2})}{\Gamma(\frac{n}{2})\Gamma(\frac{1}{2})},r\text{为偶数}\\
            0,r\text{为奇数}
        \end{cases}$$
        特别地，$n\ge 2$时,$\E[T]=0$；$n\ge 3$时，$\Var(T)=\dfrac{n}{n-2}$
        \item $t_1$就是柯西分布：$t_1(x)=\dfrac{1}{\pi(1+x^2)}$
        \item $\lim_{n\to\infty}t_n=N(0,1)$.
    \end{enumerate}
\end{proposition}

\subsection{F分布}

\begin{definition}
    设$\rv X\sim \chi_m^2,Y\sim\chi_n^2$且两者独立，则称
    $$F=\frac{X/m}{Y/n}$$
    是自由度为m和n的\textbf{F变量}，其分布函数称为自由度为m和n的\textbf{F分布}，记为$F\sim F_{m,n}$.
\end{definition}

\begin{theorem}
    $\rv Z\sim F_{m,n}$的概率密度函数为
    $$f_{m,n}(x)=B\left(\frac{m}{2},\frac{n}{2}\right)m^{\frac{m}{2}}n^{\frac{n}{2}}x^{\frac{m}{2}-1}(n+mx)^{-\frac{m+n}{2}},x>0$$
\end{theorem}

自由度m,n有序；$f_{m,n}(x)$是偏态的，m一定,n越小，偏态越严重。

将$F_{m,n}$分布的上侧$\alpha(0<\alpha<1)$分位数记为$c=F_{m,n}(\alpha)$，它满足$F\sim F_{m,n},\mathbb{P}(F>c)=\alpha$。

\begin{proposition}[F变量的性质]
    设$Z\sim F_{m,n}$，则\begin{enumerate}
        \item $1/Z\sim F_{n,m}$
        \item $\forall 0<r<\dfrac{n}{2}$，有$$\E[Z^r]=\left(\frac{n}{m}\right)^r\frac{\Gamma(\frac{m}{2}+r)\Gamma(\frac{n}{2}-r)}{\Gamma(\frac{n}{2})\Gamma(\frac{m}{2})}$$
        特别地，$$\E[Z]=\frac{n}{n-2},n>2,\quad \Var(Z)=\frac{2n^2(m+n-2)}{m(n-2)^2(n-4)},n>4$$
        \item 若$T\sim t_m$，则$T^2\sim F_{1,m}$
        \item $F_{m,n}(1-\alpha)=\dfrac{1}{F_{n,m}(\alpha)}$
    \end{enumerate}
\end{proposition}

\section{次序统计量的分布}

\subsection{次序统计量的分布}

\begin{lemma}
    设总体$X$的分布函数为$F(x)$，密度函数为$f(x)$，$X_1,\cdots,X_n$为简单样本，则有
    $$\idotsint_{a<x_1<\cdots<x_n<b}f(x_1)\cdots f(x_n)dx_1\cdots dx_n=\frac{1}{n!}[F(b)-F(a)]^n,-\infty\le a<b\le \infty$$
\end{lemma}

\begin{theorem}[次序统计量的分布]
    设$X_1,\cdots,X_n\iid\sim F$，有概率密度函数$f$，$(X_{(1)},\cdots,X_{(n)})$为次序统计量。记$Y_i=X_{(i)}$，则次序统计量的联合密度为
    $$g(y_1,\cdots,y_n)=\begin{cases}
        n!f(y_1)\cdots f(y_n),&\text{if }y_1<\cdots<y_n\\
        0,&\text{otherwise}
    \end{cases}$$
\end{theorem}
\begin{proof}
    $\Supp g$显然是$y_1<\cdots<y_n$.每个$g(y_1,\cdots,y_n)$对应$n!$种排列方式导致的$n!$种情况。
\end{proof}

\begin{theorem}[两个次序统计量的联合分布]
    设$X_1,\cdots,X_n\iid\sim F$，有概率密度函数$f$，$(X_{(1)},\cdots,X_{(n)})$为次序统计量，则其中任意二者$(X,Y)=(X_{(i)},X_{(j)})(i<j)$的联合密度为
    $$f_{ij}(x,y)=\begin{cases}
        \dfrac{n!f(x)f(y)}{(i-1)!(j-i-1)!(n-j)!}F^{i-1}(x)(F(y)-F(x))^{j-i-1}(1-F(y))^{n-j},&\text{if }x<y\\
        0,&\text{otherwise}
    \end{cases}$$
\end{theorem}
\begin{proof}
    $f_{ij}(x,y)$是$g(y_1,\cdots,y_n)$的边缘分布。将积分分离变量如下：
    \begin{align*}
        f_{ij}(x,y)=&n!f(x)f(y)\idotsint_{-\infty<x_1<\cdots<x_{i-1}<x}f(x_1)\cdots f(x_{i-1})\,dx_1\cdots dx_{i-1}\\
        &\times \idotsint_{x<x_{i+1}<\cdots<x_{j-1}<y}f(x_{i+1})\cdots f(x_{j-1})\,dx_{i+1}\cdots dx_{j-1}\\
        &\times\idotsint_{y<x_{j+1}<\cdots<x_n<\infty}f(x_{j+1})\cdots f(x_n)\,dx_{j+1}\cdots dx_n
    \end{align*}
    利用lemma2.3.1，得到\begin{align*}
        f_{ij}(x,y)=n!f(x)f(y)\cdot\frac{F^{i-1}(x)}{(i-1)!}\cdot\frac{[F(y)-F(x)]}{(j-i-1)!}\cdot\frac{[1-F(y)]^{n-j}}{(n-j)!}
    \end{align*}
\end{proof}

类似可求任意个次序统计量的联合分布。

\begin{theorem}[单个次序统计量的分布]
    设$X_1,\cdots,X_n\iid\sim F$，有概率密度函数$f$，$(X_{(1)},\cdots,X_{(n)})$为次序统计量，则$X=X_{(m)}$的概率密度函数为
    $$f_m(x)=m\binom{n}{m}F^{m-1}(x)[1-F(x)]^{n-m}f(x)$$
    分布函数为
    $$F_m(x)=m\binom{n}{m}\int_{0}^{F(x)}t^{m-1}(1-t)^{n-m}\,dt$$
    特别地，令$m=1$得到样本极小值的概率密度函数：
    $$f_1(x)=n(1-F(x))^{n-1}f(x)$$
    令$m=n$得到样本极大值的概率密度函数：
    $$f_n(x)=nF^{n-1}(x)f(x)$$
\end{theorem}

\subsection{极差的分布}

极差$R=X_{(n)}-X_{(1)}$.令\begin{align*}
    \begin{cases}
        Z=X_{(1)}\\
        R=X_{(n)}-X_{(1)}
    \end{cases}\iff\begin{cases}
        X_{(1)}=Z\\
        X_{(n)}=Z+R
    \end{cases}
\end{align*}

则雅可比行列式
$$J=\left|\frac{\partial(X_{(1)},X_{(n)})}{\partial(Z,R)}\right|=1$$
因此\begin{align*}
    g(z,r)&=Jf_{ij}(z,z+r)\\
    &=\begin{cases}
        n(n-1)[F(z+r)-F(z)]^{n-2}f(z+r)f(z),&r>0\\
        0,&r\le 0
    \end{cases}
\end{align*}

求边缘分布即得$R$的概率密度函数为
$$g_R(r)=\int_{-\infty}^{\infty}g(z,r)\,dz$$

\subsection{均匀分布情形}

令$X_1,\cdots,X_n\iid\sim U(0,1)$，则
$$f_m(x)=m\binom{n}{m}x^{m-1}(1-x)^{n-m}I_{(0,1)}(x)$$
$$f_{ij}(x,y)=\begin{cases}
    \dfrac{n!}{(i-1)!(j-i-1)!(1-y)!}x^{i-1}(y-x)^{j-i-1}(1-y)^{n-j},&0<x<y<1\\
    0,&\text{otherwise}
\end{cases}$$
$(X_{(1)},\cdots,X_{(n)})$的联合概率密度函数为
$$f(x_{(1)},\cdots,x_{(n)})=n!,0<x_{(1)}<\cdots<x_{(n)}<1$$
极差的概率密度函数为
$$g_R(r)=n(n-1)r^{n-2}(1-r)I_{(0,1)}$$

\section{统计量的极限分布}

\begin{definition}[极限分布/大样本分布]
    如果样本容量$n\to\infty$时的统计量分布趋于一确定分布，则该分布称为统计量的\textbf{极限分布/大样本分布}。
\end{definition}
\begin{definition}[大样本性质/小样本性质]
    样本容量$n\to\infty$时的统计量或统计推断方法的性质称为\textbf{大样本性质}；当样本容量固定时称为\textbf{小样本性质}。    
\end{definition}

\begin{example}[强相合性]
    设$X_1,\cdots,X_n\iid\sim F$，其中总体$F$均值为$a$，方差为$\sigma^2<\infty$，样本均值为
    $$\bar{X}_n=\frac{1}{n}\sum_{i=1}^n X_i$$
    由Kolmogorov强大数定律，有
    $$\mathbb{P} (\lim_{n\to\infty}\bar{X}_n=a)=1\iff \bar{X}_n\xrightarrow{\aee}a$$
    该性质称为$\bar{X}_n$的\textbf{强相合性}，为大样本性质。
\end{example}

\begin{example}
    假设同上，由Lindeberg中心极限定理，
    $$\frac{\sqrt{n}(\bar{X}_n-a)}{\sigma}\xrightarrow{\mathcal{L} }N(0,1)$$
    其中$\xrightarrow{\mathcal{L} }$表示依分布收敛。该性质称为$\bar{X}_n$的\textbf{渐近正态性}，为大样本性质。
\end{example}

\begin{example}
    假设同上，$\E[\bar{X}_n]=a$称为$\bar{X}_n$的\textbf{无偏性}，为小样本性质。
\end{example}

\section{充分统计量}

\textbf{充分性原则}：如果$T(\mathbf{X})$为参数$\theta$的充分统计量，则依赖样本$\mathbf{X}$对$\theta$做的任何统计推断都可以仅通过统计量$T(\mathbf{X})$进行，而不会损失关于$\theta$的信息。

\begin{definition}[充分统计量]
    设样本$\mathbf{X}$的分布族为$\{f(\mathbf{x},\theta),\theta\in\Theta\}$。如果已知统计量$T=T(\mathbf{X})$的条件下，样本$\mathbf{X}$的条件分布与参数$\theta$无关，即$\forall\text{事件}A,\mathbb{P}_{\theta}(\mathbf{X}\in A|T=t)$与$\theta$无关，则称$T(\mathbf{X})$为$\theta$的\textbf{充分统计量}(sufficient statistic)。
\end{definition}
\begin{remark}
    这是充分统计量不依赖于信息论的等价定义形式，相当于说$\theta\to T(\mathbf{X})\to\mathbf{X}$构成Markov链。
\end{remark}

以下定理给出了较为简便的充分统计量判则：
\begin{theorem}[因子分解定理]
    设样本$\mathbf{X}=(X_1,\cdots,X_n)\sim f(\mathbf{x},\theta)$统计量$T(\mathbf{X})$是充分统计量$\iff f(\mathbf{x},\theta)$可分解为以下形式：
    $$f(\mathbf{x},\theta)=g(\mathbf{T(x)},\theta)h(\mathbf{x})$$
\end{theorem}
直观：给定$\mathbf{t(x)}$，$f(\mathbf{x},\theta)$就可以由$h(\mathbf{x})$归一化获得。

\begin{corollary}
    设$\mathbf{T}=\mathbf{T}(\mathbf{X})$是$\theta$的充分统计量，$S=\varphi(\mathbf{T})$是可逆函数，则$S=\varphi(\mathbf{T})$也是$\theta$的充分统计量。
\end{corollary}
可逆函数表明$S\rightleftarrows T$，它们一定含有相同的信息。

\begin{definition}[极小充分统计量]
    关于分布族$\{f_{\theta}(x)\}$的充分统计量T(X)是其他任何充分统计量$S(X)$的函数，则称$T(X)$为关于$\{f_{\theta}(x)\}$的\textbf{极小充分统计量}(minimal sufficient statistic)
    
    定义蕴含$\theta \to S(X) \to T(X) \to X$
\end{definition}

\begin{theorem}
    设样本$\mathbf{X}=(X_1,\cdots,X_n)$的概率密度函数为$f(\mathbf{x},\boldsymbol{\theta}),x\in\mathcal{X} ,\boldsymbol{\theta}\in\boldsymbol{\Theta}$.对于充分统计量$\mathbf{T}(\mathbf{x})$，如果对任意两个样本点$\mathbf{x},\mathbf{y}$，
    $$\frac{f(\mathbf{x},\boldsymbol{\theta})}{f(\mathbf{y},\boldsymbol{\theta})}\text{为与}\theta\text{无关的常数}\iff \mathbf{T}(\mathbf{x})=\mathbf{T}(\mathbf{y})$$
    则$\mathbf{T}(\mathbf{x})$是$\boldsymbol{\theta}$的极小充分统计量。
\end{theorem}
\begin{proof}
    要证：任一其他充分统计量$\mathbf{T}^*(\mathbf{X})$，$\mathbf{T}(\mathbf{X})$都是$\mathbf{T}^*(\mathbf{X})$的函数。

    根据银子分解定理，存在$g^*,h^*$使得
    $$f(\mathbf{x},\theta)=g^*(\mathbf{T^*(x)},\theta)h^*(\mathbf{x})$$
    令样本点$\mathbf{x},\mathbf{y}\in\mathcal{X}$满足$\mathbf{T}^*(\mathbf{x})=\mathbf{T}^*(\mathbf{y})$，则
    $$\frac{f(\mathbf{x},\theta)}{f(\mathbf{y},\theta)}=\frac{g^*(\mathbf{T}^*(\mathbf{x}),\theta)h^*(\mathbf{x})}{g^*(\mathbf{T}^*(\mathbf{y}),\theta)h^*(\mathbf{y})}=\frac{h^*(\mathbf{x})}{h^*(\mathbf{y})}$$
    与$\theta$无关，因此$\mathbf{T}(\mathbf{x})=\mathbf{T}(\mathbf{y})$，也就是说
    $$\mathbf{T}^*(\mathbf{x})=\mathbf{T}^*(\mathbf{y})\implies \mathbf{T}(\mathbf{x})=\mathbf{T}(\mathbf{y})$$
    因此以下函数是良定义的：
    $$\varphi:\mathbf{T}^*(\mathbf{x})\mapsto\mathbf{T}(\mathbf{x}),\forall \mathbf{x}\in\mathcal{X}$$
\end{proof}
\begin{remark}
    我们只用到了
    $$\frac{f(\mathbf{x},\boldsymbol{\theta})}{f(\mathbf{y},\boldsymbol{\theta})}\text{为与}\theta\text{无关的常数}\implies \mathbf{T}(\mathbf{x})=\mathbf{T}(\mathbf{y})$$
    但这不影响定理表述的正确性，因为根据因此分解定理，$\impliedby$是自动成立的。
\end{remark}

\begin{proposition}
    根据指数族的性质可以得到：
    \begin{enumerate}
        \item 二项分布$B(m,\theta)$中，$\theta$的极小充分统计量为$\sum_{i=1}^n X_i$或$\bar{X}$
        \item Poisson分布$P(\lambda)$中，$\lambda$的极小充分统计量为$\sum_{i=1}^n X_i$或$\bar{X}$
        \item 正态分布$N(\mu,\sigma^2)$中，$\mathbf{T}(\mathbf{X})=\left(\sum_{i=1}^n X_i,\sum_{i=1}^n X_i^2\right)$或$\mathbf{T}(\mathbf{X})=\left(\bar{X},S^2\right)$为$(\mu,\sigma^2)$的极小充分统计量
        \item Gamma分布$\Gamma(r,\lambda)$中，$\mathbf{T}(\mathbf{X})=\left(\sum_{i=1}^n X_i,\sum_{i=1}^n \ln(X_i)\right)$为$(r,\lambda)$的极小充分统计量
    \end{enumerate}
\end{proposition}

\section{点估计与估计优劣的评价}

\begin{definition}[点估计]
    设$\mathbf{X}=(X_1,\cdots,X_n)$为从某个总体$F_\theta$中抽取的样本，$g(\theta)$为总体参数$\theta$的实函数，用$\hat{g}(\mathbf{X})$作为$g(\theta)$的\textbf{估计量}(estimator)，称为\textbf{点估计}(point estimation)，在一组具体的样本值$\mathbf{x}=(x_1,\cdots,x_n)$下，$\hat{g}(\mathbf{x})$称为$g(\theta)$的\textbf{估计值}。
\end{definition}

\subsection{无偏性}

\begin{definition}[无偏估计]
    设$\mathbf{X}=(X_1,\cdots,X_n)$为总体$\{F_\theta,\theta\in\Theta\}$中抽取的样本，$\hat{g}_n(\mathbf{X})$是$g(\theta)$的一个估计量。如果
    $$\E_\theta[\hat{g}_n(\mathbf{X})]=g(\theta),\theta\in\Theta$$
    则称$\hat{g}_n(\mathbf{X})$是$g(\theta)$的一个\textbf{无偏估计}(unbiased estimation)；如果
    $$\lim_{n\to\infty}\E_\theta[\hat{g}_n(\mathbf{X})]=g(\theta),\theta\in\Theta$$
    则称$\hat{g}_n(\mathbf{X})$是$g(\theta)$的一个\textbf{渐进无偏估计}(asymptotically unbiased estimation).
\end{definition}

无偏性要求\textbf{估计量大量使用}，根据大数定律，计算很多次$\hat{g}_n(\mathbf{X})$后，$\hat{g}_n(\mathbf{X})$的均值以概率1等于$g(\theta)$.如果只统计一次，无偏性就没有用。

\begin{example}
    设$\mathbf{X}=(X_1,\cdots,X_n)$为均值为$\mu$，方差为$\sigma^2$的总体中抽取的样本，则样本方差$S^2$是$\sigma^2$的无偏估计，因为
    \begin{align*}
        \E[S^2]=&\frac{1}{n-1}\left(\sum_{i=1}^n \E[X_i^2]-n\E[\bar{X}^2]\right)\\
        =&\frac{n}{n-1}(\E[X_1^2]-\E[\bar{X}^2])\\
        =&\frac{n}{n-1}\left[(\sigma^2+\mu^2)-\left(\frac{\sigma^2}{n}+\mu^2\right)\right]\\
        =&\sigma^2
    \end{align*}
\end{example}

\subsection{有效性}

无偏估计量，方差越小，有效性越高：

\begin{definition}[有效性]
    设$\hat{g}_1(\mathbf{X})$和$\hat{g}_2(\mathbf{X})$是$g(\theta)$的两个不同的无偏估计量，若
    $$\Var_\theta(\hat{g}_1(\mathbf{X}))\le \Var_\theta(\hat{g}_2(\mathbf{X}))$$
    且至少存在一个$\theta$使严格不等号成立，则称估计量$\hat{g}_1(\mathbf{X})$比$\hat{g}_2(\mathbf{X})$\textbf{有效}。
\end{definition}

\subsection{相合性}

\begin{definition}[相合性]
    设$\hat{g}_n(\mathbf{X})=\hat{g}_n(X_1,\cdots,X_n)$是$g(\theta)$的一个估计量，如果$\hat{g}_n(\mathbf{X})$依概率收敛到$g(\theta)$，即
    $$\lim_{n\to\infty}\mathbb{P}_\theta(|\hat{g}_n(\mathbf{X})-g(\theta)|\ge \epsilon)=0,\forall \theta\in\Theta,\forall \epsilon>0$$
    则称$\hat{g}_n(\mathbf{X})$为$g(\theta)$的\textbf{弱相合估计}(weakly consistent estimation).
    
    如果$\hat{g}_n(\mathbf{X})\aee$收敛到$g(\theta)$，即
    $$\mathbb{P}_\theta(\lim_{n\to\infty}\hat{g}_n(\mathbf{X})=g(\theta))=1,\forall \theta\in\Theta$$
    则称$\hat{g}_n(\mathbf{X})$为$g(\theta)$的\textbf{强相合估计}(strongly consistent estimation).

    如果对$r>0$，有
    $$\lim_{n\to\infty}\E_\theta|\hat{g}_n(\mathbf{X})-g(\theta)|^r=0,\forall\theta\in\Theta$$
    则称$\hat{g}_n(\mathbf{X})$为$g(\theta)$的\textbf{r阶矩相合估计}(consistent estimation in r'th maen).当$r=2$时称为\textbf{均方相合估计}(consistent estimation in quadratic maen)。
\end{definition}

相合性是估计量的大样本性质。

强相合$\implies$弱相合；r阶矩相合$\implies$弱相合。

\section{矩估计}

核心思想：\textbf{样本矩估计总体矩}

设$X_1,\cdots,X_n$是从总体$F$抽取的简单样本，则样本的$k$阶原点矩
$$a_{nk}=\frac{1}{n}\sum_{i=1}^n X_i^k,\quad k=1,2,\cdots$$
是总体$k$阶原点矩$\alpha_k=\E[X_k]$的\textbf{矩估计量}，为无偏估计；

样本的$k$阶中心矩
$$m_{nk}=\frac{1}{n}\sum_{i=1}^n (X_i-\bar{X})^k,\quad k=2,3,\cdots$$
是总体$k$阶中心矩$\mu_k=\E[X-\E[X]]^k$的\textbf{矩估计量}，不是无偏估计，而是系统地偏低。例如$S^2=\frac{n}{n-1}m_{n2}$是$\sigma^2$的无偏估计。

\begin{definition}[矩估计]
    设$g(\theta)$可以表示为总体分布的某些矩的函数：
    $$g(\theta)=G(\alpha_1,\cdots,\alpha_k;\mu_2,\cdots,\mu_s)$$
    设$\mathbf{X}$是从分布族中抽取的简单样本，将矩用矩估计量代替，得
    $$\hat{g}(\mathbf{X})=G(a_{n1},\cdots,a_{nk};m_{n2},\cdots,m_{ns})$$
    称$\hat{g}(\mathbf{X})$是$g(\theta)$的\textbf{矩估计量}(moment estimate)，这种求矩估计量的方法称为\textbf{矩法}。
\end{definition}

\section{最大似然估计}

\begin{definition}[似然函数]
    取定$\theta$，$f(\mathbf{x},\theta)$为样本$\mathbf{X}=(X_1,\cdots,X_n)$的概率函数，固定$\mathbf{x}$，作为$\theta$的函数，称为\textbf{似然函数}(likelihood function)，记为
    $$L(\theta,\mathbf{x})=f(\mathbf{x},\theta)$$
    将$l(\theta,\mathbf{x})=\ln L(\theta,\mathbf{x})$称为\textbf{对数似然函数}。
\end{definition}

\begin{definition}[最大似然估计]
    设$\mathbf{X}=(X_1,\cdots,X_n)$是从参数分布族$\mathcal{F}=\{f(\mathbf{x},\theta):\theta\in\Theta\}$中抽取的简单样本，$L(\theta,\mathbf{x})$是似然函数，若存在统计量$\hat{\theta}=\hat{\theta}(\mathbf{X})$，使得
    $$L(\hat{\theta}(\mathbf{x}),\mathbf{x})=\max_{\theta\in\Theta}L(\theta,\mathbf{x}),\forall x\in\mathcal{X} $$
    则称$\hat{\theta}(\mathbf{X})$是$\theta$的\textbf{最大似然估计}(maximum likelihood estimation,MLE).若待估计函数是$g(\theta)$，则定义$g(\hat{\theta})$为$g(\theta)$的MLE.
\end{definition}

很多时候寻找对数似然函数的最大值更为容易。

\begin{example}
    设$\mathbf{X}=(X_1,\cdots,X_n)$是从两点分布族$\{B(1,p):0<p<1\}$中抽取的简单样本，求$p$的MLE.

    似然函数为
    $$L(p,\mathbf{x})=\prod_{i=1}^n (p^{x_i}(1-p)^{1-x_i})=p^{\sum_{i=1}^n x_i}(1-p)^{n-\sum_{i=1}^n x_i}$$
    对数似然函数为
    $$l(p,\mathbf{x})=\ln L(p,\mathbf{x})=\ln p\cdot \sum_{i=1}^n x_i+\ln(1-p)\cdot\left(n-\sum_{i=1}^n x_i\right)$$
    对数似然方程为
    $$\frac{\partial l(p,\mathbf{x})}{\partial p}=\frac{1}{p}\sum_{i=1}^n x_i-\frac{1}{1-p}\left(n-\sum_{i=1}^n x_i\right)$$
    求解并将$x$换成$X$得估计量
    $$\hat{p}=\frac{1}{n}\sum_{i=1}^n X_i=\bar{X}$$

    当$\bar{X}\in(0,1)=\Theta$时，$p=0,1\notin\Theta$，但由于$0,1$是$\Theta$的极限点，可以补充MLE的定义并认为$p=0,1$是MLE.
\end{example}

\begin{example}
    设$\mathbf{X}=(X_1,\cdots,X_n)$是从正态分布族$\{N(\mu,\sigma^2):\mu\in\mathbb{R},\sigma^2>0\}$中抽取的简单样本，求$(\mu,\sigma^2)$的MLE.

    样本$\mathbf{X}$的分布为
    $$f(\mathbf{x},\boldsymbol{\theta})=(2\pi\sigma^2)^{-\frac{n}{2}}\exp\left[-\frac{1}{2\sigma^2}\sum_{i=1}^n (x_i-\mu)^2\right]$$
    对数似然函数为
    $$l(\boldsymbol{\theta},\mathbf{x})=-\frac{n}{2}\ln(2\pi)-n\ln\sigma-\frac{1}{2\sigma^2}\sum_{i=1}^n (x_i-\mu)^2$$
    对数似然方程组为
    \begin{align*}
        \frac{\partial l(\boldsymbol{\theta},\mathbf{x})}{\partial \mu}&=\frac{1}{\sigma^2}\sum_{i=1}^n (x_i-\mu)=0\\
         \frac{\partial l(\boldsymbol{\theta},\mathbf{x})}{\partial \sigma^2}&=-\frac{n}{2\sigma^2}+\frac{1}{2\sigma^4}\sum_{i=1}^n (x_i-\mu)^2=0
    \end{align*}
    解得
    $$\hat{\mu}=\frac{1}{n}=\bar{X},\quad \hat{\sigma^2}=\frac{1}{n}\sum_{i=1}^n(X_i-\hat{X})^2=S_n^2$$
    MLE未必具有无偏性。
\end{example}

\subsection{最大似然估计的性质}

\begin{theorem}
    设$\mathbf{X}=(X_1,\cdots,X_n)$是从总体$\{f(x,\theta):\theta\in\Theta\}$中抽取的简单样本，若$T=T(\mathbf{X})$是$\theta$的充分统计量，且$\theta$的最大似然估计$\hat{\theta}$唯一存在，则它必为$T$的函数。
\end{theorem}
\begin{proof}
    由因子分解定理，
    $$L(\theta,x)=g(T(\mathbf{x}),\theta)h(\mathbf{x})$$
    记$\theta$的唯一最大似然估计为$\hat{\theta}$，则
    $$L(\hat{\theta},\mathbf{x})=\max_{\theta\in\Theta}L(\theta,\mathbf{x})\iff g(T(\mathbf{x}),\hat{\theta})=\max_{\theta\in\Theta}g(T(\mathbf{x}),\theta)$$
    而使$g(T(\mathbf{x}),\theta)$达到上确界的$\theta$必为$T$的函数。
\end{proof}
该定理表明，如果最大似然估计$\hat{\theta}$是充分统计量，则$\hat{\theta}$是极小充分统计量。

\section{Bayes点估计}

\subsection{定义}

\begin{definition}[Bayes估计]
    Bayes估计有三种，都可以记为$\hat{\theta}_B$：\begin{enumerate}
        \item \textbf{最大后验概率估计}(Maximum A Posteriori, MAP)/\textbf{后验众数估计}(posterior mode estimation)/\textbf{广义最大似然估计}(广义MLE)：$$\hat{\theta}_{MD}=\arg\max_{\theta\in\Theta}\pi(\theta|\mathbf{x})$$
        \item \textbf{后验中位数估计}(posterior median estimation)：用后验分布的中位数$\hat{\theta}_{ME}$作为$\theta$的估计量
        \item \textbf{后验期望估计}(posterior expectation estimation)：用后验分布的期望值$\hat{\theta}_{E}$作为$\theta$的估计量
    \end{enumerate}
\end{definition}

当后验密度单峰对称时，上述三种Bayes估计相等；当先验分布为共轭先验时，Bayes估计的计算比较容易。

\subsection{精度}

经典估计中希望估计量的\textbf{均方误差}(MSE)最小，应用到Bayes估计：

\begin{definition}[后验均方误差]
    设$\theta$的后验分布为$\pi(\theta|\mathbf{x})$，Bayes估计为$\hat{\theta}_B(\mathbf{x})$，定义其\textbf{后验均方误差}(posterior mean sqare error,PMSE)为
    $$\PMSE(\hat{\theta}_B(\mathbf{x}))=\E^{\theta|\mathbf{x}}[(\theta-\hat{\theta}_B)^2]$$
    PMSE衡量估计精度，越小越好。
\end{definition}

特别地，采用后验期望估计时，PMSE恰取到最小值，且最小值为后验方差，即后验期望估计是PMSE最小准则下的唯一最优估计：
$$\PMSE(\hat{\theta}_B(\mathbf{x}))=\E^{\theta|\mathbf{x}}[(\theta-\hat{\theta}_B(\mathbf{x}))]=\E^{\theta|\mathbf{x}}[(\theta-\hat{\theta}_E)+(\hat{\theta}_E-\hat{\theta}_B)]^2=\Var^\pi(\mathbf{x})+(\hat{\theta}_E-\hat{\theta}_B)^2\ge \Var^\pi(\mathbf{x})$$
其中$\Var^\pi(\mathbf{x})$表示后验方差。

\chapter{区间估计}

\section{区间估计的基本概念}

\subsection{参数的区间估计问题}

\begin{definition}[区间估计]
    设有参数分布族$\mathcal{F}=\{f(x,\theta):\theta\in\Theta\}$，$g(\theta)$是定义在参数空间$\Theta$上的已知函数，$\mathbf{X}=(X_1,\cdots,X_n)$是从分布族$\mathcal{F}$中抽取的简单样本，统计量$\hat{g}_1(\mathbf{X}),\hat{g}_2(\mathbf{X}):\mathcal{X}\to\Theta$满足$\hat{g}_1(\mathbf{X})\le\hat{g}_2(\mathbf{X})$，则称随机区间$[\hat{g}_1(\mathbf{X}),\hat{g}_2(\mathbf{X})]$为$g(\theta)$的一个\textbf{区间估计}(interval estimation).
\end{definition}

衡量区间估计的指标为\textbf{可靠度}与\textbf{精度}。样本大小一定，可靠性与精度不可能同时变大。通常的方案是在保证一定可靠度的前提下尽可能提高精度。随机区间的平均长度$\E_\theta(\hat{\theta}_2-\hat{\theta}_1)$是一种典型的精度指标。

\subsection{置信区间}

简便起见以下设$g(\theta)=\theta$.

\begin{definition}[置信水平与置信系数]
    设随机区间$[\hat{\theta}_1,\hat{\theta}_2]$为参数$\theta$的一个区间估计，称
    $$\mathbb{P}_\theta(\hat{\theta}_1\le \theta\le \hat{\theta}_2)$$
    为次区间估计的\textbf{置信水平}(confidence level).将置信水平在参数空间上的下确界：
    $$\inf_{\theta\in\Theta}\mathbb{P}_\theta(\hat{\theta}_1\le \theta\le \hat{\theta}_2)$$
    称为该区间估计的\textbf{置信系数}(confidence coeffient).
\end{definition}

置信系数给出了可靠性的下确界。

\begin{definition}[置信区间]
    设$[\hat{\theta}_1(\mathbf{X}),\hat{\theta}_2(\mathbf{X})]$是$\theta$的一个区间估计，若对于给定的$0<\alpha<1$，有
    $$\mathbb{P}_\theta\left(\hat{\theta}_1(\mathbf{X})\le \theta\le\hat{\theta}_2(\mathbf{X})\right)\ge 1-\alpha,\quad\theta\in\Theta$$
    则称$[\hat{\theta}_1(\mathbf{X}),\hat{\theta}_2(\mathbf{X})]$是$\theta$的\textbf{置信水平}为$1-\alpha$的\textbf{置信区间}(confidence interval)，将
    $$\mathbb{P}_\theta\left(\hat{\theta}_1(\mathbf{X})\le \theta\le\hat{\theta}_2(\mathbf{X})\right)$$
    称为$[\hat{\theta}_1(\mathbf{X}),\hat{\theta}_2(\mathbf{X})]$
    的\textbf{置信系数}。
\end{definition}

\subsection{置信限}

\begin{definition}[置信限]
    设统计量$\hat{\theta}_U(\mathbf{X})$和$\hat{\theta}_L(\mathbf{X})$对给定的$0<\alpha<1$，有
    $$\mathbb{P}_\theta(\theta\le \hat{\theta}_U(\mathbf{X}))\ge 1-\alpha,\forall\theta\in\Theta$$
    $$\mathbb{P}_\theta(\hat{\theta}_L(\mathbf{X})\le\theta)\ge 1-\alpha,\forall\theta\in\Theta$$
    则称$\hat{\theta}_U(\mathbf{X})$和$\hat{\theta}_L(\mathbf{X})$分别是$\theta$的置信水平为$1-\alpha$的（单侧）\textbf{置信上限}(upper confidence limit)和\textbf{置信下限}(lower confidence limit).
    $$\inf_{\theta\in\Theta}\mathbb{P}_\theta(\theta\le \hat{\theta}_U(\mathbf{X})),\quad\inf_{\theta\in\Theta}\mathbb{P}_\theta(\hat{\theta}_L(\mathbf{X})\le\theta)$$
    分别成为置信上、下限的置信系数。
\end{definition}

\begin{lemma}
    设$\hat{\theta}_U(\mathbf{X})$和$\hat{\theta}_L(\mathbf{X})$分别是$\theta$的置信水平为$1-\alpha_1$和$1-\alpha_2$的单侧置信下限和置信上限，且对任何样本都有$\hat{\theta}_U(\mathbf{X})\le\hat{\theta}_L(\mathbf{X})$，则$[\hat{\theta}_U(\mathbf{X}),\hat{\theta}_L(\mathbf{X})]$是$\theta$的置信水平为$1-(\alpha_1+\alpha_2)$的双侧置信区间。
\end{lemma}

\subsection{置信域}

\begin{definition}[置信域]
    设有参数分布族$\mathcal{F}=\{f(x,\boldsymbol{\theta}):\boldsymbol{\theta}\in\boldsymbol{\Theta}\}$，参数空间$\boldsymbol{\Theta}\in\mathbb{R}^k,k\ge 2$，$\mathbf{X}=(X_1,\cdots,X_n)$是来自分布族$\mathcal{F}$的样本，若统计量$S(\mathbf{X})$满足\begin{enumerate}
        \item $\forall\mathbf{X}\in\mathcal{X},S(\mathbf{X})\subset\boldsymbol{\Theta}$
        \item $\forall 0<\alpha<1,\mathbb{P}_{\boldsymbol{\theta}}(\boldsymbol{\theta}\in S(\mathbf{X}))\ge 1-\alpha,\forall\boldsymbol{\theta}\in\boldsymbol{\Theta}$
    \end{enumerate}
    则称$S(\mathbf{X})$是$\boldsymbol{\theta}$的置信水平为$1-\alpha$的\textbf{置信域}(confidence region)或\textbf{置信集}，将
    $$\inf_{\theta\in\Theta}\mathbb{P}_{\boldsymbol{\theta}}(\boldsymbol{\theta}\in S(\mathbf{X}))$$
    称为置信系数。
\end{definition}

\chapter{参数假设检验}

\section{假设检验的若干基本概念}

\begin{definition}[统计假设和假设检验]
    对总体分布的某些特征所作的假设称为\textbf{统计假设}；使用样本对所作出的统计假设进行推断的方法和过程称为\textbf{假设检验}。根据总体分布的形式是否一致，假设检验问题分为两大类：\textbf{参数型假设检验}和\textbf{非参数型假设检验}
\end{definition}

\begin{definition}[原假设与备择假设]
    设有参数分布族$\{f(x,\theta):\theta\in\Theta\}$，$X_1,\cdots,X_n$是从中抽取的简单样本，对于参数空间的子集$\Theta_0\subset\Theta$，如果我们对于$H_0:\theta\in\Theta_0$有较大的把握，就将命题$H_0:\theta\in\Theta_0$称为\textbf{零假设}(null hypothesis)或\textbf{原假设}；设$\Theta_1\subset\Theta-\Theta_0$，则命题$H_1:\theta\in\Theta_1$称为$H_0$的\textbf{备择假设}(alternative hypothesis)或\textbf{对立假设}。将上述问题写为：
    $$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$

    如果$\Theta_0$或$\Theta_1$只包含一个点，则称对应假设是\textbf{简单假设}(simple hypothesis)，否则称为是\textbf{复合假设}(composite hypothesis).
\end{definition}

\begin{definition}[检验统计量，拒绝域]
    给定检验问题
    $$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$
    根据\textbf{检验统计量}$T(\mathbf{X})$进行判决，对于区域$D\in\mathcal{X}$，当$T(\mathbf{X})\in D$时拒绝$H_0$，反之接受$H_0$，称$D$为\textbf{拒绝域}(reject region)或\textbf{否定域}，将$\bar{D}=\mathcal{X}-D$称为\textbf{接受域}(acceptance region).

    我们称法则
    $$T:\begin{cases}
        T(\mathbf{X})\in D&,\text{拒绝}H_0\\
        T(\mathbf{X})\in\bar{D}&,\text{接受}H_0
    \end{cases}$$
    是上述检验问题的一个\textbf{检验}(test)。
\end{definition}

\begin{definition}[检验函数]
    为便于数学上的处理，引入\textbf{检验函数}$\varphi:\mathcal{X} \to[0,1]$，它与检验$T$是一一对应的：
    $$\varphi(\mathbf{x})=\begin{cases}
        1,&\mathbf{x}\in D\\
        0,&\mathbf{x}\in \bar{D}
    \end{cases}$$
    它表示有了样本$\mathbf{X}$后，拒绝$H_0$的概率。
    
    如果$\varphi(\mathbf{x})$只取值于$\{0,1\}$，称这种检验为\textbf{非随机化检验}(non-randomized test)；如果对某些样本$\mathbf{X}$有$0<\varphi(\mathbf{x})<1$，则称之为\textbf{随机化检验}(randomized text)，这种情况发生时，可以根据一个成功概率为$r$的随机试验决定是否接受，此时随机化检验函数具有以下形式：
    $$\varphi(\mathbf{x})=\begin{cases}
        1,&\mathbf{x}\in D\\
        r,&\mathbf{x}\notin D\cup \bar{D}\\
        0,&\mathbf{x}\in \bar{D}
    \end{cases}$$
\end{definition}

\begin{definition}[功效函数]
    假设检验问题
    $$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$
    中可能出现的两类错误：\begin{enumerate}
        \item 零假设$H_0$本来是对的，由于样本的随机性，观察值落入拒绝域$D$，错误地将$H_0$拒绝，称为\textbf{弃真}或\textbf{第一类错误}
        \item 零假设$H_0$本来是错的，由于样本的随机性，观察值落入接受域$\bar{D}$，错误地将$H_0$接受，称为\textbf{取伪}或\textbf{第二类错误}
    \end{enumerate}
    为了计算犯两类错误的概率，引入检验函数$\varphi$的\textbf{功效函数}(power function)（或称效函数、势函数）：
    $$\beta_\varphi(\theta)=\E_\theta[\varphi(\mathbf{X})],\quad \theta\in\Theta$$
\end{definition}

功效函数表示当参数为$\theta$时，拒绝$H_0$的概率。我们希望功效函数$\beta_\varphi(\theta)$在$\Theta_0$中尽可能小而在$\Theta_1$中尽可能大。

\begin{definition}[检验水平]
    如果检验$\varphi$犯第一类错误的概率不超过$\alpha$，即
    $$\beta_\varphi(\theta)\le \alpha,\quad\forall\theta\in\Theta_0$$
    则称$\alpha$是$\varphi$的一个水平，将$\varphi$称为\textbf{显著性水平}为$\alpha$的检验，简称水平为$\alpha$的检验。
\end{definition}

调整接受域时，两类错误概率必然一增大一减小。Neyman-Pearson原则：在保证第一类错误概率不超过某个小的数$\alpha$的前提下，寻找第二类错误概率较小的检验。此时$H_0$受保护，因此通常设为不能被轻易拒绝的命题。我们将这样的假设检验问题称为\textbf{显著性检验}(significant test)，其检验水平$\alpha$称为\textbf{显著性水平}。

按定义，显著性水平可以取的更大，为此称一个检验的最小水平为其\textbf{真实水平}，但在造表时也将$\alpha$取为0.01，0.05等典型值。

水平$\alpha$选的越小，接受原假设$H_0$的结论可靠性就越低，而拒绝原假设$H_0$的可靠性就越高。

\begin{example}
    某工厂生产一批产品要上市销售，按规定次品率$p$不得超过0.01，今在其中随机不放回抽取100件，根据样本中的次品率决定能否出厂。

    选取\textbf{零假设}和\textbf{备择假设}如下：
    $$H_0:0<p\le 0.01\leftrightarrow H_1:0.01<p<1$$
    这两个假设都是复合假设。

    定义
    $$X_i=\begin{cases}
        1,&\text{第i个产品为次品}\\
        0,&\text{otherwise}
    \end{cases}$$
    将\textbf{检验统计量}选为$T(\mathbf{X})=\sum_{i=1}^{100} X_i$，\textbf{否定域}选为$D=\{\mathbf{X}=(X_1,\cdots,X_n):T(\mathbf{X})>c\}$，如果选取接受域为$\bar{D}=\{\mathbf{X}=(X_1,\cdots,X_n):T(\mathbf{X})\le c\}$，则为\textbf{非随机化检验}，\textbf{检验函数}为
    $$\varphi(\mathbf{x})=\begin{cases}
        1,&T(\mathbf{x})>c\\
        0,&T(\mathbf{x})\le c
    \end{cases}$$
    也可以认为$c$为\textbf{临界值}(critical value)，因为当$T(\mathbf{x})$恰等于$c$时，厂方和买方倾向于做出相反的判断，此时可以取随机化检验函数为
    $$\varphi(\mathbf{x})=\begin{cases}
        1,&T(\mathbf{x})>c\\
        0.5,&T(\mathbf{x})=c\\
        0,&T(\mathbf{x})<c
    \end{cases}$$
    这样可以通过抛一枚均匀硬币来决定是否接受$H_0$.

    如果总体的真实次品率$p<0.01$，但由于样本的随机性，抽样结果显示$T(\mathbf{x})>c$，则犯第一类错误；如果总体的真实次品率$p>0.01$，但由于样本的随机性，抽样结果显示$T(\mathbf{x})<c$，则犯第二类错误。

    利用$T(\mathbf{X})=\sum_{i=1}^{100} X_i\sim B(n,\theta),0<\theta<1$，可知（非随机化）检验的\textbf{功效函数}为
    \begin{align*}
        \beta_\varphi(\theta)&=\E_\theta[\varphi(\mathbf{X})]\\
        &=\mathbb{P}_\theta\left(\sum_{i=1}^{100}X_i>c\right)\\
        &=\sum_{k=c+1}^{100}\binom{100}{k}\theta^k(1-\theta)^{100-k}
    \end{align*}
\end{example}

\section{似然比检验}

对于检验问题
$$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$
比较拒绝域内外的最高似然水平：
$$L_{\Theta_0}(\mathbf{x})=\sup_{\theta\in\Theta_0}f(\mathbf{x},\theta)$$
$$L_{\Theta_1}(\mathbf{x})=\sup_{\theta\in\Theta_1}f(\mathbf{x},\theta)$$
考虑比值$L_{\Theta_1}(\mathbf{x})/L_{\Theta_0}(\mathbf{x})$，比值越小，说明$\theta\in\Theta_1$的似然性越小，越应当倾向于接受$H_0$.

为了计算上的方便，我们注意到
$$\begin{cases}
    L_{\Theta_0}(\mathbf{x})>L_{\Theta_1}(\mathbf{x})&\implies L_{\Theta}(\mathbf{x})=L_{\Theta_0}(\mathbf{x})\\
    L_{\Theta_0}(\mathbf{x})<L_{\Theta_1}(\mathbf{x})&\implies L_{\Theta}(\mathbf{x})=L_{\Theta_1}(\mathbf{x})
\end{cases}$$
于是
$$\lambda(\mathbf{x})=\frac{L_{\Theta}(\mathbf{x})}{L_{\Theta_0}(\mathbf{x})}=\max\left\{\frac{L_{\Theta_1}(\mathbf{x})}{L_{\Theta_0}(\mathbf{x})},1\right\}$$
与上述比值$L_{\Theta_1}(\mathbf{x})/L_{\Theta_0}(\mathbf{x})$同增同减，可以由此进行似然比检验：

\begin{definition}[似然比检验]
    设样本$\mathbf{X}$有概率函数$f(\mathbf{x},\theta),\theta\in\Theta$，$\Theta_0$为参数空间$\Theta$的真子集，考虑检验问题
    $$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$
    则统计量
    $$\lambda(\mathbf{x})=\frac{\sup_{\theta\in\Theta}f(\mathbf{x},\theta)}{\sup_{\theta\in\Theta_0}f(\mathbf{x},\theta)}$$
    称为关于该检验问题的\textbf{似然比}。定义\textbf{检验函数}为
    $$\varphi(\mathbf{x})=\begin{cases}
        1&,\lambda(\mathbf{x})>c\\
        r&,\lambda(\mathbf{x})=c\\
        0&,\lambda(\mathbf{x})<c
    \end{cases}$$
    称之为上述检验问题的一个\textbf{似然比检验}(likelihood ratio test,LRT)，其中$c,r(0\le r\le 1)$为待定常数。
\end{definition}

参数$c,r$应使检验具有给定水平$\alpha$，这要求我们求出$\lambda(\mathbf{x})$的分布并根据$\alpha$选取参数$c,r$，但$\lambda(\mathbf{x})$的分布通常难以求得。对此我们有两种策略，一种是用$\lambda(\mathbf{x})$的极限分布近似代替，另一种使寻找统计量$T(\mathbf{x})$使得$\lambda(\mathbf{x})=g(T(\mathbf{x}))$为$T(\mathbf{x})$的严格递增（或严格递减）函数，这样，似然比检验等价于
$$\varphi(\mathbf{x})=\begin{cases}
        1&,T(\mathbf{x})>c'\\
        r&,T(\mathbf{x})=c'\\
        0&,T(\mathbf{x})<c'
    \end{cases}$$
因此只需要求$T(\mathbf{x})$的分布。

\section{Bayes假设检验和预测}

对于检验问题
$$H_0:\theta\in\Theta_0\leftrightarrow H_1:\theta\in\Theta_1$$
其中$\Theta=\Theta_0\sqcup\Theta_1$.在Bayes统计方法中，获得后验分布$\pi(\theta|\mathbf{x})$后，计算假设$H_0$和和$H_1$的后验概率$$\alpha_0=\mathbb{P}(\Theta_0|\mathbf{x}),\quad\alpha_1=\mathbb{P}(\Theta_1|\mathbf{x})$$
当\textbf{后验机会比（后验概率比）}$\alpha_0/\alpha_1$显著小于1时接受$H_1$，显著大于1是接受$H_0$，如果约等于1则需要进一步抽样或获取先验信息。

Bayes假设检验更简单，不需要选择检验统计量、确定抽样分布、检验水平，$H_0,H_1$地位平等，容易推广到多重假设检验。

\subsection{Bayes因子}

\begin{definition}[Bayes因子]
    设两个假设$\Theta_0,\Theta_1$的先验概率分别为$\pi_1,\pi_2$，，后验概率分别为$\alpha_0,\alpha_1$，比例$\alpha_0/\alpha_1$称为$H_0$对$H_1$的\textbf{后验机会比}(posterior odds ratio)，$\pi_1/\pi_2$被称为\textbf{先验机会比}(prior odds ratio)，则\textbf{Bayes因子}定义为
    $$B_{01}^\pi(\mathbf{x})=\frac{\text{后验机会比}}{\text{先验机会比}}=\frac{\alpha_0/\alpha_1}{\pi_1/\pi_2}=\frac{\alpha_0\pi_1}{\alpha_1\pi_0}$$
    $B_{01}^\pi(\mathbf{x})\in(0,\infty)$的值越大，对$H_0$的支持程度越高。
\end{definition}

\begin{example}[简单假设情形]
    设
    $$H_0:\theta\in\Theta_0=\{\theta_0\}\leftrightarrow H_1:\theta\in\Theta_1=\{\theta_1\}$$
    则
    $$\alpha_0=\mathbb{P}(\theta_0|\mathbf{x})=\frac{f(\mathbf{x}|\theta_0)\pi_0}{f(\mathbf{x}|\theta_0)\pi_0+f(\mathbf{x}|\theta_1)\pi_1}$$
    $$\alpha_1=\mathbb{P}(\theta_1|\mathbf{x})=\frac{f(\mathbf{x}|\theta_1)\pi_1}{f(\mathbf{x}|\theta_0)\pi_0+f(\mathbf{x}|\theta_1)\pi_1}$$
    后验机会比为
    $$\alpha_0/\alpha_1=\frac{\pi_0 f(\mathbf{x}|\theta_0)}{\pi_1 f(\mathbf{x}|\theta_1)}$$
    Bayes因子为
    $$B_{01}^\pi(\mathbf{x})=\frac{\alpha_0\pi_1}{\alpha_1\pi_0}=\frac{f(\mathbf{x}|\theta_0)}{f(\mathbf{x}|\theta_1)}$$
    是$\Theta_0\leftrightarrow\Theta_1$的似然比，不依赖于先验分布，因此\textbf{Bayes因子可以看作是数据$\mathbf{x}$支持$\Theta_0$的程度}。
    
    拒绝假设$H_0$要求$\alpha_0/\alpha_1<1$，即
    $$B_{01}^\pi(\mathbf{x})=\frac{f(\mathbf{x}|\theta_0)}{f(\mathbf{x}|\theta_1)}>\frac{\pi_0}{\pi_1}$$
    相当于一种似然比检验。
\end{example}

在复杂假设情形下，$B_{01}^\pi(\mathbf{x})$是加权似然比，与$\Theta_0,\Theta_1$上的先验概率密度$g_0,g_1$有关，只能部分消除先验分布的影响。

\subsection{多重假设检验}

\begin{definition}[多重假设检验]
    \textbf{多重假设检验}(multiple hypothesis testing)问题：
    $$H_i:\theta\in\Theta_i,i=1,\cdots,k,\quad\sqcup_{i=1}^k\Theta_i=\Theta$$
    计算后验概率
    $$\alpha_i=\mathbb{P}(\Theta_i|\mathbf{x}),i=1,\cdots,k$$
    若$\alpha_{i_0}$最大，则接受假设$H_{i_0}$.
\end{definition}

\subsection{预测问题}

对随机变量的未来观测值做统计推断称为\textbf{预测}。预测有两种情形：\begin{enumerate}
    \item 设$X\sim f(x|\theta)$，令$\mathbf{X}=(X_1,\cdots,X_n)$为从总体$X$中获得的历史数据，要对随机变量$X$的未来观测值$X_0$做出推断
    \item 设$X\sim f(x|\theta)$，令$\mathbf{X}=(X_1,\cdots,X_n)$为从总体$X$中获得的历史数据，要对具有密度函数$g(z|\theta)$的随机变量$Z$的未来观测值$Z_0$做出推断（通常假定$X,Z$不相关）
\end{enumerate}

1是2的特例。

\textbf{Bayes预测}：以$\pi(\theta|\mathbf{x})$表示参数$\theta$的后验密度，并认为给定参数$\theta$后，未来观测值$Z$与历史数据$\mathbf{X}$条件独立，即
$$g(z|\theta,\mathbf{x})=g(z|\theta)$$
因此给定历史数据$\mathbf{X}$时$(Z.\theta)$的联合分布为
$$g(z|\theta)\pi(\theta|\mathbf{x})$$
对$\theta$积分就得到给定$\mathbf{X}$条件下$Z$的条件边缘密度，以它作为预测密度：
\begin{definition}[后验预测密度]
    设参数$\theta$的后验密度是$\pi(\theta|\mathbf{x})$，给定$\mathbf{x}$，随机变量$Z$的\textbf{后验预测密度}(posterior predictive density)定义为
    $$p(z|\mathbf{x})=\int_{\Theta}g(z|\theta)\pi(\theta|\mathbf{x})\,d\theta$$
\end{definition}

\chapter{非参数假设检验}

\section{符号检验和符号秩和检验}

\subsection{符号检验法}

\begin{example}
    为比较甲乙两种酒的优劣，请$N$个人品尝并给出对两种酒的评分，分别记为$X_i,Y_i,i=1,\cdots,N$.记$Z_i=X_i-Y_i$.

    如果假定$Z_i\sim N(\mu,\sigma^2)$，则原问题转化为参数假设检验：
    $$H_0:\mu=0\leftrightarrow H_1:\mu\neq 0$$

    如果不能假定$Z_i$服从正态分布，则可以对每个人的评分给出一个符号
    $$S_i=\begin{cases}
        +,&\ Z_i>0\\
        -,&\ Z_i<0\\
        0,&\ Z_i=0
    \end{cases}$$
    非参数假设检验问题
    $$H_1:\text{甲乙两种酒一样好}\leftrightarrow H_2:\text{甲乙两种酒不一样}$$
    基于符号$S_1,\cdots,S_N$，故称为\textbf{符号检验}(sign test).
\end{example}

符号检验实际上是二项分布参数检验的一个特例。记$+,-$的数量分别为$n_+,n_-$，有
$$n_+ +n_- =n,\quad n_+\sim B(n,\theta)$$
上述检验问题转化为
$$H_1:\theta=\frac{1}{2}\leftrightarrow H_2:\theta\neq\frac{1}{2}$$

\subsection{符号秩和检验}

\begin{definition}
    设$X_1,\cdots,X_n$为两两不相等的一组样本，将其按大小排列为
    $X_{(1)}<\cdots<X_{(n)}$，称第$R_i$小的样本$X_i$在样本$(X_1,\cdots,X_n)$的\textbf{秩}为$R_i$，即$X_i=X_{(R_i)}$.
\end{definition}

\begin{definition}
    设$X_1,\cdots,X_n$为来自单个总体的样本，或来自多个总体样本的合样本，则$\mathbf{R}=(R_1,\cdots,R_n)$称为$(X_1,\cdots,X_n)$的\textbf{秩统计量}(rank statistics)，其中$R_i$为$X_i$的秩。

    由$\mathbf{R}$导出的统计量也称为秩统计量。基于秩统计量的检验方法统称为\textbf{秩检验}(rank test)。
\end{definition}

\chapter{统计判决理论}

\section{基本概念}

\subsection{统计判决三要素，Bayes统计判决第四要素}

如果做出决策时所依据的事实中至少有一部分时受到随机性影响的观察值，则可称之为统计判决问题。

\subsubsection{样本空间和样本分布族}

样本空间$\mathcal{X}$；取值于$\mathcal{X}$的随机变量$X$及其分布族$\{f(x|\theta),\theta\in\Theta\}$.其中$\theta$为未知参数。

\subsubsection{行动空间}

决策者对某个统计判决问题可能采取的行动所构成的非空集合称为\textbf{行动空间}(action space)，记为$\mathcal{A}$.

例如，在估计问题中，$\mathcal{A}=\Theta$；在检验问题中，$\mathcal{A}=\{\text{接受原假设}H_0,\text{拒绝原假设}H_0\}$.

\subsubsection{损失函数}

\textbf{损失函数}(loss function)是定义于$\mathcal{A}\times \Theta$上的非负函数$L(a,\theta)$，表示参数为$\theta$时，采取行动$a\in\mathcal{A}$所承受的损失。

\subsubsection{先验分布}

定义在参数空间$\Theta$上的先验分布$\pi(\theta)$

统计判决问题研究如何根据样本$\mathbf{x}$的值恰当选取行动$a$使得按样本分布或后验分布计算的平均损失最小。

\subsection{统计判决理论的一些基本概念}

\begin{definition}[决策函数]
    函数$\delta:\mathcal{X}\to\mathcal{A}$称为\textbf{决策函数}(decision function)或\textbf{判决函数}，表示根据样本$\mathbf{x}$采取的行动$\delta=\delta(\mathbf{x})\in\mathcal{A}$。
\end{definition}

采取行动$\delta$时的损失依赖于参数$\theta$，因此可以将该行动造成的损失描述为函数$L(\delta,\theta)$，使用取定参数$\theta$时的统计平均作为$\delta$的衡量标准，就引入以下风险函数的概念：

\begin{definition}[风险函数]
    设$\delta$是一个决策函数，称平均损失
    $$R(\delta,\theta)=\E^{\mathbf{X}|\theta}[L(\delta(\mathbf{X}),\theta)]=\int_{\mathcal{X}}L(\delta(\mathbf{x}),\theta)f(\mathbf{x}|\theta)\,d\mathbf{x}$$
    为$\delta$的\textbf{风险函数}(risk funtion).
\end{definition}

A.Wald：评价一个决策函数的唯一依据就是其风险函数，风险函数越小越好，即：

\begin{definition}[一致最优决策函数]
    记$R(\delta,\theta)$是决策函数$\delta$的风险函数，若存在一个决策函数$\delta^*$，使得$\forall \delta$有
    $$R(\delta^*,\theta)\le R(\delta,\theta),\forall\theta\in\Theta$$
    则称$\delta^*$为\textbf{一致最优解}或\textbf{一致最优决策函数}。
\end{definition}

通常较难找到一致最优决策函数，此时可以制定另外的优良性准则，或者对决策函数提出合理要求，缩小所考虑的决策函数的范围。

\subsection{Bayes统计判决的一些基本概念}

Bayes准则下的决策函数优良性通过风险函数对参数的先验分布的统计平均来衡量：

\begin{definition}[Bayes风险]
    设$R(\delta,\theta)$是决策函数$\delta$的风险函数，$\pi(\theta)$为$\theta$的先验分布，则称
    $$R_\pi(\delta)=\E^\theta[R(\delta,\theta)]=\begin{cases}
        \int_{\Theta}R(\delta,\theta)\pi(\theta)\,d\theta,&\text{当}\theta\text{为连续型随机变量}\\
        \sum_{i}R(\delta,\theta_i)\pi(\theta_i),&\text{当}\theta\text{为离散型随机变量}
    \end{cases}$$
    为$\delta$的\textbf{Bayes风险}(Bayes risk).
\end{definition}

希望最小化Bayes风险：

\begin{definition}[Bayes解]
    设$\delta_1$和$\delta_2$分别为$\theta$的两个决策函数，若$R_\pi(\delta_1)\le R_\pi(\delta_2)$，则称$\delta_1$在Bayes风险的意义下优于$\delta_2$。

    在Bayes风险的意义下的最优决策函数$\delta^*$满足
    $$R_\pi(\delta^*)\le R_\pi(\delta),\forall \delta$$
    称$\delta^*$为该统计判决问题的\textbf{Bayes解}。
\end{definition}

如果将前面讨论的先验分布改成后验分布，我们可以得到一套类似的理论：

\begin{definition}[后验风险]
    设$\pi(\theta|\mathbf{x})$为$\theta$的后验分布，$L(\delta,\theta)$为损失函数，则称
    $$R(\delta|\mathbf{x})=\E^{\theta|\mathbf{x}}[R(\delta,\theta)]=\begin{cases}
        \int_{\Theta}R(\delta(\mathbf{x}),\theta)\pi(\theta|\mathbf{x})\,d\theta,&\text{当}\theta\text{为连续型随机变量}\\
        \sum_{i}R(\delta(\mathbf{x}),\theta_i)\pi(\theta_i|\mathbf{x}),&\text{当}\theta\text{为离散型随机变量}
    \end{cases}$$
\end{definition}

\begin{definition}[最优Bayes决策函数]
    若存在决策函数$\delta^*$，使得
    $$R(\delta^*|\mathbf{x})=\min_\delta R(\delta|\mathbf{x})$$
    则称$\delta^*$为\textbf{在后验风险最小准则下最优的Bayes决策函数}。
\end{definition}

\begin{proposition}
    后验风险与Bayes风险的关系如下：
    $$R_\pi(\delta)=\E^{\mathbf{X}}[R(\delta|\mathbf{X})]$$
\end{proposition}
\begin{proof}
    利用
    $$f(\mathbf{x},\theta)=f(\mathbf{x}|\theta)\pi(\theta)=\pi(\theta|\mathbf{x})m(\mathbf{x})$$
    （其中$m(\mathbf{x})$为$\mathbf{x}$的边缘分布），我们有
    \begin{align*}
        R_\pi(\delta)=&\E^\theta[R(\delta,\theta)]=\int_{\Theta}R(\delta,\theta)\pi(\theta)\,d\theta\\
        =&\int_{\Theta}\left[\int_{\mathcal{X}}L(\delta,\theta)f(\mathbf{x}|\theta)\,d\mathbf{x}\right]\pi(\theta)\,\theta\\
        =&\int_{\mathcal{X}}\left[\int_{\Theta}L(\delta,\theta)\,d\theta\right]m(\mathbf{x})\,d\mathbf{x}\\
        =&\int_{\mathcal{X}}R(\delta|\mathbf{x})m(\mathbf{x})\,d\mathbf{x}\\
        =&m(\mathbf{x})
    \end{align*}
\end{proof}

总之，Bayes风险
$$R(\delta|\mathbf{x})=\E^{\theta|\mathbf{x}}[R(\delta,\theta)]=\E^{\mathbf{X}}[R(\delta|\mathbf{X})]$$
这表明它即可以看作风险函数关于$\theta$的先验分布的平均，也可看作后验风险关于$\mathbf{X}$的边缘分布的平均。
